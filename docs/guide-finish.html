add one adapter

blocks.registerAdapter("block-counter", {
  mount({ host, settings }) {
    const blockCounterButton = document.createElement("button");
    blockCounterButton.textContent = settings.start;
    host.appendChild(blockCounterButton);
    return blockCounterButton;
  }
});

blocks.register({
  id: "block-click-counter",
  label: "click counter",
  adapter: "block-counter",
  medium: "html",
  defaults: { start: 0 }
});

const blockCounterHost = document.createElement("div");
const blockCounter = blocks.add(blockCounterHost, {
  id: "block-counter"
});
blockCounter.menu("counter");

await blocks.mount(
  "block-click-counter",
  blockCounterHost,
  { start: 3 }
);

finish

  • Keep renderer-specific code in the adapter, outside the core.
  • Run npm run check.
